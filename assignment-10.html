Q1. Personalized Login Greeting

Displays a greeting according to the current hour.

const userName = prompt("Enter your name:") || "Guest";
const currentHour = new Date().getHours();
let greeting;

if (currentHour < 12) {
  greeting = `Good Morning ${userName}!`;
} else if (currentHour <= 17) {
  greeting = `Good Afternoon ${userName}!`;
} else {
  greeting = `Good Evening ${userName}!`;
}

console.log(greeting);

Q3. Monthly Expense Tracker

Calculates total, average, and total after adding 10% tax.

const expenses = {
  food: 4500,
  travel: 1200,
  rent: 8000,
  bills: 1500,
  leisure: 1000
};

let total = 0;
for (const amount of Object.values(expenses)) {
  total += amount;
}
const average = total / Object.keys(expenses).length;
total += total * 0.10;

console.log(`Total before tax: ₹${(total / 1.10).toFixed(2)}`);
console.log(`Average expense: ₹${average.toFixed(2)}`);
console.log(`Final amount after 10% tax: ₹${total.toFixed(2)}`);

Q4. Academic Performance Evaluator

Checks the average percentage and automatically detains a student if any subject mark is below 35.

const marks = [88, 76, 91, 82, 79];
const total = marks.reduce((sum, mark) => sum + mark, 0);
const average = total / marks.length;
const percentage = (total / (marks.length * 100)) * 100;
const failedSubject = marks.some(mark => mark < 35);
let result;

if (failedSubject) {
  result = "Detained";
} else if (percentage >= 85 && average >= 85) {
  result = "Promoted with Distinction";
} else if (percentage >= 50 && percentage < 85) {
  result = "Promoted";
} else {
  result = "Detained";
}

console.log(`Average: ${average.toFixed(2)}`);
console.log(`Percentage: ${percentage.toFixed(2)}%`);
console.log(`Result: ${result}`);

Q7. Smart Guessing Game (Number Range)

Generates a secret number from 1–50 and checks a test guess.

const secretNumber = Math.floor(Math.random() * 50) + 1;
const userGuess = 25; // Change this test value if you want
let message;

if (userGuess === secretNumber) {
  message = "Correct guess!";
} else {
  if (Math.abs(userGuess - secretNumber) <= 3) {
    message = "Very close!";
  } else if (userGuess > secretNumber && userGuess >= 1 && userGuess <= 50) {
    message = "Too high";
  } else if (userGuess < secretNumber && userGuess >= 1 && userGuess <= 50) {
    message = "Too low";
  } else {
    message = "Guess must be between 1 and 50.";
  }
}

console.log(`Secret number: ${secretNumber}`);
console.log(`Your guess: ${userGuess}`);

Q8. Employee Salary Projection

Projects salary for five years using a loop and yearly increment rate.

let salary = 40000;
const incrementRate = 8;
const rows = [];

for (let year = 1; year <= 5; year++) {
  salary += salary * incrementRate / 100;
  salary = Math.round(salary);
  rows.push({ year: `Year ${year}`, salary: `₹${salary.toLocaleString("en-IN")}` });
}

Q9. Random Math Quiz Generator

Creates a random arithmetic question and calculates the answer using switch.

const number1 = Math.floor(Math.random() * 20) + 1;
const number2 = Math.floor(Math.random() * 20) + 1;
const operators = ["+", "-", "*", "/"];
const operator = operators[Math.floor(Math.random() * operators.length)];
let answer;

switch (operator) {
  case "+": answer = number1 + number2; break;
  case "-": answer = number1 - number2; break;
  case "*": answer = number1 * number2; break;
  case "/": answer = (number1 / number2).toFixed(2); break;
}

console.log(`Question: ${number1} ${operator} ${number2} = ?`);
console.log(`Correct answer: ${answer}`);
