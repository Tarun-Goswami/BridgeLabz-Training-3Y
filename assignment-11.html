Q1. Scope Conflict Resolver

Uses a global bonus and a local salary/permanent-status variable.

let bonus = 5000;

function calculateSalary(isPermanent) {
  let salary = 40000;
  if (isPermanent) {
    salary += bonus;
  }
  console.log(`Total salary: ₹${salary}`);
  return salary;
}

calculateSalary(true);
calculateSalary(false);
console.log(`Global bonus remains: ₹${bonus}`);

Q4. Array Performance Analyzer

Creates eight random scores and calculates highest, lowest, average, and passing students.

const scores = Array.from({ length: 8 },
  () => Math.floor(Math.random() * 71) + 30);
const highest = Math.max(...scores);
const lowest = Math.min(...scores);
const total = scores.reduce((sum, score) => sum + score, 0);
const average = total / scores.length;
const passed = scores.filter(score => score >= 50).length;
const labels = scores.map((score, index) => `Student ${index + 1}: ${score}`);

console.log(`Highest: ${highest}`);
console.log(`Lowest: ${lowest}`);
console.log(`Average: ${average.toFixed(2)}`);

Q6. Event-Based Counter Simulation

Increment and decrement functions update a shared counter; a nested function reads the updated value.

let count = 0;

function showCount() {
  function display() {
    console.log(`Current count: ${count}`);
    return count;
  }
  return display();
}

function increment() {
  count++;
  return showCount();
}

function decrement() {
  count--;
  return showCount();
}

increment();
increment();
decrement();

Q8. Dynamic Discount Evaluator

Applies category discounts and an additional 5% discount if the original cart total exceeds ₹50,000.

const cart = [
  { item: "Laptop", category: "electronics", price: 45000 },
  { item: "Shoes", category: "fashion", price: 2500 },
  { item: "Book", category: "education", price: 600 }
];

const subtotal = cart.reduce((sum, product) => sum + product.price, 0);
let discountedTotal = 0;

for (const product of cart) {
  let discountRate = 0;
  if (product.category === "electronics") discountRate = 0.10;
  else if (product.category === "fashion") discountRate = 0.05;
  discountedTotal += product.price * (1 - discountRate);
}

if (subtotal > 50000) {
  discountedTotal *= 0.95;
}

console.log(`Original cart total: ₹${subtotal.toFixed(2)}`);
console.log(`Final cart total: ₹${discountedTotal.toFixed(2)}`);
Output will appear here.

Q9. Odd–Even Number Analyzer

Checks each number from 1 to 30. FizzBuzz takes priority when a number is divisible by both 3 and 5.

const results = [];

for (let number = 1; number <= 30; number++) {
  let label;
  if (number % 3 === 0 && number % 5 === 0) {
    label = "FizzBuzz";
  } else if (number % 2 === 0) {
    label = "Even";
  } else {
    label = "Odd";
  }
  results.push(`${number}: ${label}`);
}

console.log(results);
Output will appear here.

Q10. Departmental Employee Evaluator

Assigns a performance label to each department using nested if-else statements.

const departments = [
  ["HR", 72],
  ["Finance", 88],
  ["Tech", 95],
  ["Support", 63]
];

for (const department of departments) {
  const name = department[0];
  const score = department[1];
  let performance;

  if (score >= 90) performance = "Excellent";
  else if (score >= 75) performance = "Good";
  else if (score >= 60) performance = "Average";
  else performance = "Needs Improvement";

  console.log(`${name}: ${score} — ${performance}`);
}
